def get_trPL(data_trPL, trPL_lyr):
    ''' Process the output of the tj file to extract the trPL data for the specified layer, taking into account the blocking layers and the layer indexing.

    Parameters
    ----------
    data_trPL : DataFrame
        The data from the tj file, which contains the time (t) and the current density (Jdir) for each layer 
    trPL_lyr : int
        The index of the layer to calculate the trPL for, starting from 1 for the first layer of the simulation setup with blocking layers 
    
    Returns
    -------
    DataFrame
        A DataFrame containing the time (t) and the trPL signal (trPL) for the specified layer, where trPL is equal to the current density (Jdir) for that layer. 
    '''

    # Only a single layer currently
    trPL = data_trPL[f'JdirL{trPL_lyr}'] # We use the direct recombination current density (Jdir) for the trPL signal.
    t = data_trPL['t']

    # save the trPL data to a file
    df_trPL = pd.DataFrame({'t': t, 'trPL': trPL})

    return df_trPL

def plot_trPL(session_path, output_file, xlim = None, ylim = None, xscale = 'log', yscale = 'log'):
    ''' Simple plot of the trPL data from the output file generated by the run_trPL_simu function. 
    The output file should contain two columns: time (t) and trPL signal (trPL). The function will create a plot of trPL vs time on a log-log scale.

    Parameters
    ----------
    session_path : string
        The path to the session folder, which is the working directory for the simulation
    output_file : string
        The name of the output file containing the trPL data, which should be located in the session path
    xlim : list, optional
        The limits for the x-axis, by default None, which means that the limits will be determined automatically based on the data
    ylim : list, optional
        The limits for the y-axis, by default None, which means that the limits will be determined automatically based on the data
    xscale : string, optional
        The scale for the x-axis, by default 'log'
    yscale : string, optional
        The scale for the y-axis, by default 'log'

    Returns
    -------
    None

    '''
    # simple trPL plot
    data_trPL = pd.read_csv(os.path.join(session_path, output_file),sep=r'\s+')
    t = data_trPL['t']*1E9 # Convert time to ns for plotting
    trPL = data_trPL['trPL']

    fig, ax = plt.subplots()
    ax.plot(t, trPL)
    ax.set_xlabel('Time [ns]')
    ax.set_ylabel(r'trPL | Jdir [Am$^{-2}$]')
    ax.set_title('Time Resolved Photoluminescence (trPL) Simulation')
    ax.set_xscale(xscale)
    ax.set_yscale(yscale)

    if xlim is not None:
        xlim = [x*1E9 for x in xlim]
        ax.set_xlim(xlim)
    if ylim is not None:
        ax.set_ylim(ylim)

    plt.tight_layout()
    plt.show()
    return

def plot_trPL_pulse(session_path, output_file, freq, t_pulse, pulse = -1, xlim = None, ylim = None, xscale = 'log', yscale = 'log'):
    ''' Plot the trPL signal of one or several pulses from a pulse train simulation (run_trPL_simu with num_pulses and freq).
    Pulse n (starting from 1) covers the period from (n-1)/freq to n/freq and is centred at t_pulse + (n-1)/freq. 
    The time axis of each pulse is shifted to start at 0 at the start of its period, so several pulses can be compared directly on the same plot.

    The number of pulses in the file is the number of pulse centres (t_pulse + (n-1)/freq) within the simulated time range. 
    This can be smaller than num_pulses, as ZimT stops the simulation early when the state at the end of a pulse no longer 
    changes from pulse to pulse. If the simulation was interrupted (e.g. timeout), the last pulse can be incomplete.

    Parameters
    ----------
    session_path : string
        The path to the session folder, which is the working directory for the simulation
    output_file : string
        The name of the output file containing the trPL data, which should be located in the session path
    freq : float
        The repetition frequency of the pulses in Hz, as used in the simulation
    t_pulse : float
        The center of the pulse within each period in s, as used in the simulation
    pulse : int or list of int, optional
        The pulse(s) to plot, starting from 1 for the first pulse. Negative values count from the end of the file,
        i.e. -1 is the last simulated pulse. Use a list (e.g. [1, 2, -1]) to plot several pulses on top of each other, by default -1
    xlim : list, optional
        The limits for the x-axis in s (relative to the start of the pulse), by default None
    ylim : list, optional
        The limits for the y-axis, by default None
    xscale : string, optional
        The scale for the x-axis, by default 'log'
    yscale : string, optional
        The scale for the y-axis, by default 'log'

    Returns
    -------
    None

    '''
    data_trPL = pd.read_csv(os.path.join(session_path, output_file),sep=r'\s+')
    t = data_trPL['t'].values
    t_period = 1/freq

    # Tolerance to compare time values, as the time values in the file have a limited precision: half of the smallest time step
    dt = np.diff(t)
    tol = 0.5*dt[dt > 0].min() if np.any(dt > 0) else 0

    # Number of pulses in the file: number of pulse centres t_pulse + n/freq (n = 0, 1, ...) within the simulated time range
    num_pulses = int(np.floor((t.max() - t_pulse + tol)*freq)) + 1
    if num_pulses < 1:
        raise ValueError(f'The output file ends at t = {t.max():.3e} s, before the center of the first pulse (t_pulse = {t_pulse:.3e} s).')

    pulses = [pulse] if np.isscalar(pulse) else list(pulse)

    # Convert the requested pulses (1 = first, -1 = last) to indices starting from 0
    indices = []
    for p in pulses:
        if p == 0 or abs(p) > num_pulses:
            raise ValueError(f'pulse must be between 1 and {num_pulses} or between -{num_pulses} and -1, as the file contains {num_pulses} pulse(s). Got {p}.')
        indices.append(p - 1 if p > 0 else num_pulses + p)

    fig, ax = plt.subplots()
    for idx in indices:
        # Select the period of this pulse, including both its start and end point, so all pulses have the same time grid
        t_rel = t - idx*t_period # Time relative to the start of the period of this pulse
        mask = (t_rel > -tol) & (t_rel < t_period + tol)
        ax.plot(t_rel[mask]*1E9, data_trPL['trPL'].values[mask], label=f'Pulse {idx+1}') # time in ns

    ax.set_xlabel('Time since start of pulse [ns]')
    ax.set_ylabel(r'trPL | Jdir [Am$^{-2}$]')
    if len(indices) == 1:
        ax.set_title(f'trPL simulation, pulse {indices[0]+1} of {num_pulses}')
    else:
        ax.set_title(f'trPL simulation, {num_pulses} pulse(s) in file')
        ax.legend()
    ax.set_xscale(xscale)
    ax.set_yscale(yscale)

    if xlim is not None:
        xlim = [x*1E9 for x in xlim]
        ax.set_xlim(xlim)
    if ylim is not None:
        ax.set_ylim(ylim)

    plt.tight_layout()
    plt.show()
    return

def run_trPL_simu(zimt_device_parameters, session_path, lyr_idx, t_max, t_scale, t_step, t_pulse, FWHM, genProfile='calc', spectrum_type = 'laser', I = 6E15, wavelength = 640E-9, G_ehp = 1E31, contactless_device = True, format_output=True, run_mode=False, tVG_name = 'tVG.txt', output_file = 'trPL.dat', tj_name = 'tj.dat', log_name = 'log.txt',varFile ='none', block_layer_file='Block.txt', freq = None, num_pulses = None, steps = 1000, G_frac = 1, steps_FWHM = 200, **kwargs):
    """ Run a time resolved PhotoLuminescence (trPL) simulation, requiring the definition of a (Gaussian) light pulse.
    There is an option to use a contactless device architecture, which adds blocking layers to the simulation setup. 
    The output PL is given in terms of the equivalent direct recombination current (Jdir) of the layer specified by lyr_idx.

    Parameters
    ----------
    zimt_device_parameters : string
        The name of the ZimT device parameters file, which should be located in the session path
    session_path : string
        The path to the session folder, which is the working directory for the simulation
    lyr_idx : int
        The index of the active layer to calculate the trPL for, starting from 1 for the first layer of the original setup (i.e. without the blocking layers)
    t_max : float
        The maximum time for the experiment (single pulse only)
    t_scale : string
        The time scale to use for the time array, which can be 'lin' for linear or 'log' for logarithmic
    t_step : float
        The time step (single pulse only)
    t_pulse : float
        The center of the pulse in time (within each period for a pulse train). Make sure it is far enough from t=0 otherwise first part of pulse will be cut, resulting in lower number of photons
    FWHM : float
        The full width at half maximum (FWHM) of the pulse
    genProfile : string
        The type of generation profile to use, either 'calc', 'none', or 'userFile', by default 'calc'
    spectrum_type : string
        The type of spectrum to use, either 'spectrum' or 'laser', by default 'spectrum'
    I : float
        The intensity of the pulse in photons/m^2/pulse, only used if spectrum_type is 'laser', by default 1E15
    wavelength : float
        The wavelength of the pulse in meters, only used if spectrum_type is 'laser', by default 400E-9
    G_ehp : float, optional
        The generation rate in m^-3 s^-1, only used if genProfile is 'none', by default 1E31
    contactless_device : bool, optional
        Whether to use a contactless device architecture, which adds blocking layers to the simulation setup, by default True
    format_output : bool, optional
        Whether to format the output files to remove the blocking layers and update the layer indices back to their original values, by default True
    run_mode : bool, optional
        If False, show verbose output in console, by default False
    tVG_name : string, optional
        The name of the tVG file to create, by default 'tVG.txt'
    output_file : string, optional
        The name of the output file to create with the trPL data, by default 'trPL.dat'
    tj_name : string, optional
        The name of the tj file to create, by default 'tj.dat'
    log_name : string, optional
        The name of the log file to create, by default 'log.txt'
    varFile : string, optional
        The name of the var file to create, by default 'none'. If 'none', no var file will be created and the varFile parameter will not be passed to ZimT.
    block_layer_file : string, optional
        The name of the blocking layer file, by default 'Block.txt'
    freq : float, optional
        The repetition frequency of the pulses in Hz, required when num_pulses is not None, by default None
    num_pulses : int, optional
        The number of pulses. If None, a single pulse is simulated until t_max, otherwise num_pulses pulses with frequency freq are simulated (t_max and t_step are then ignored), by default None
    steps : int, optional
        The number of time steps per pulse, i.e. per period 1/freq (pulse train only), by default 1000. For t_scale = 'lin' this is the total number of points per pulse. For t_scale = 'log' this is the number of points after the pulse; the pulse itself (from t=0 to t_pulse + 2*FWHM) is resolved with an additional fixed step of FWHM/steps_FWHM.
    G_frac : float, optional
        Scaling factor applied to the generation profile, by default 1
    steps_FWHM : int, optional
        The number of time steps per FWHM used to resolve the pulse for t_scale = 'log' (pulse train only), i.e. the time step over the pulse is FWHM/steps_FWHM, by default 200
       
    Returns
    -------
    CompletedProcess
        Output object of with returncode and console output of the simulation
    string
        Return message to display on the UI, for both success and failed
    """
    verbose = kwargs.get('verbose', False) # Check if the user wants to see the console output
    UUID = kwargs.get('UUID', '') # Check if the user wants to add a UUID to the tj file name
    cmd_pars = kwargs.get('cmd_pars', None) # Check if the user wants to add additional command line parameters
    # Check if the user wants to force the use of thread safe mode, necessary for Windows with parallel simulations
    if os.name == 'nt':
        threadsafe = kwargs.get('threadsafe', True) # Check if the user wants to force the use of threads instead of processes
    else:
        threadsafe = kwargs.get('threadsafe', False) # Check if the user wants to force the use of threads instead of processes
    
    turnoff_autoTidy = kwargs.get('turnoff_autoTidy', None) # Check if the user wants to turn off the autoTidy function in SIMsalabim
    if turnoff_autoTidy is None: 
        if not threadsafe:
            turnoff_autoTidy = True
        else:
            turnoff_autoTidy = False

    # Update the file names with the UUID
    if UUID != '':
        dum_str = f'_{UUID}'
    else:
        dum_str = ''

    # Update the filenames with the UUID
    tj_name = os.path.join(session_path, tj_name)
    output_file = os.path.join(session_path, output_file)
    tVG_name = os.path.join(session_path, tVG_name)
    log_name = os.path.join(session_path, log_name)
    if UUID != '':
        tj_file_name_base, tj_file_name_ext = os.path.splitext(tj_name)
        tj_name = tj_file_name_base + dum_str + tj_file_name_ext 
        tVG_name_base, tVG_name_ext = os.path.splitext(tVG_name)
        tVG_name = tVG_name_base + dum_str + tVG_name_ext
        log_name_base, log_name_ext = os.path.splitext(log_name)
        log_name = log_name_base + dum_str + log_name_ext
        output_file_base, output_file_ext = os.path.splitext(output_file)
        output_file = output_file_base + dum_str + output_file_ext
        if varFile != 'none':
            var_file_base, var_file_ext = os.path.splitext(varFile)
            varFile = var_file_base + dum_str + var_file_ext
            varFile = os.path.join(session_path,varFile)

    # Remove the output and tj files of a previous run, so that old results cannot be mistaken for the results of this run
    for old_file in [output_file, tj_name]:
        if os.path.isfile(old_file):
            os.remove(old_file)

    # When not using a calculated or user-defined generation profile, the generation rate must be specified in the layer parameters file. 
    # Therefore, we add the generation rate to the command line parameters to update the layer parameters file accordingly.
    if genProfile == 'none': # Add the the generation rate to the arguments
        cmd_pars.append({'par': f'l{lyr_idx}.G_ehp', 'val': str(G_ehp)})

    # Add the blocking layers to the simulation setup if a contactless device architecture is used.
    if contactless_device:
        returncode, cmd_pars, block_layer_file = create_contactless_device(session_path, zimt_device_parameters, block_layer_file=block_layer_file, cmd_pars = cmd_pars)

    # Check if the layer index specified by the user makes sense given the original device architecture (taking into account blocking layers if defined)
    dev_pars,layers = load_device_parameters(session_path, zimt_device_parameters)
    if contactless_device and len(layers)-3 < lyr_idx: # Subtract 3 for the two blocking layers and the simulation setup
        msg = f'Layer index {lyr_idx} is out of bounds for the used device architecture. The device has only {len(layers)-3} layer(s).'
        return -1, msg
    elif len(layers)-1 < lyr_idx: # Subtract 1 for the simulation setup
        msg = f'Layer index {lyr_idx} is out of bounds for the used device architecture. The device has only {len(layers)-1} layer(s).'
        return -1, msg

    # Run the ZimT simulation for a (Gaussian) light pulse. The trPL signal is then calculated from the output.
    result, message = lp.run_light_pulse_simu(zimt_device_parameters, session_path, t_max, t_scale, t_step, t_pulse, FWHM, 
                                              genProfile, spectrum_type, I, wavelength, run_mode, tVG_name, tj_name, log_name, 
                                              varFile, cmd_pars, turnoff_autoTidy, verbose, threadsafe,
                                              freq=freq, num_pulses=num_pulses, steps=steps, G_frac=G_frac, steps_FWHM=steps_FWHM)

    # 0: success, 95: finished with some non-converged points, 97: timeout.
    # With a timeout, the tj file contains the results up to the moment the simulation was stopped (e.g. only part of the pulses),
    # so we still write these (partial) results to the output file.
    if result in [0, 2, 95, 97]:
        # Check if the file exists first
        if not os.path.isfile(tj_name):
            msg = f'tj file not found: {tj_name}'
            return -1, msg

        if result == 97:
            message = str(message) + ' The output file only contains the results until the simulation was stopped, increase the timeout parameter (or use timeout = -1) to simulate the full experiment.'

        data_trPL = pd.read_csv(os.path.join(session_path, tj_name),sep=r'\s+')

        # process the output to get the trPL data, this needs to take into account the blocking layers and also which layers should be taken into account for the signal
        if contactless_device:
            trPL_lyr = lyr_idx + 1
        else:
            trPL_lyr = lyr_idx

        # Calcuate the trPL signal for the specified layer
        df_trPL = get_trPL(data_trPL, trPL_lyr)
        # Make sure the precision of the output is sufficient to distinguish subsequent time values. With a pulse train, the time steps
        # in the later pulses can be many orders of magnitude smaller than the time itself (e.g. 1E-11 s steps at t = 1E-3 s).
        dt = np.diff(df_trPL['t'].values)
        dt_min = dt[dt > 0].min() if np.any(dt > 0) else 0
        t_end = df_trPL['t'].abs().max()
        if dt_min > 0 and t_end > 0:
            num_digits = max(3, int(np.ceil(np.log10(t_end/dt_min))) + 2)
        else:
            num_digits = 3
        df_trPL.to_csv(os.path.join(session_path,output_file), index=False, sep = ' ', float_format=f'%.{num_digits}e') # Write to the output file

    # Format the output files to remove the blocking layers and update the layer indices back to their original values
    if contactless_device and format_output:
        process_output_contactless(session_path, zimt_device_parameters, 'zimt', format_output = format_output, clean_simulation_setup=False, tJ_file = tj_name, Var_file = varFile, block_layer_file = block_layer_file)

    return result, message

    ########### Generation profile ###########

    # 'calc' : let SIMsalabim calculate the generation profile using the transfer matrix method

    # 'userFile' : User defined profile (replace 'userFile' with the filename ( + path) with respect to the session path). See SIMsalabim documentation for specifics on the file format and content
    genProfile = 'calc'

    # If genProfile is 'calc', specify the spectrum
    spectrum_type = 'laser' # 'spectrum' or 'laser' ( 'spectrum' will use the spectrum as specified in the simulation setup, 'laser' uses a monochromatic pulse (with no background illumination) where the wavelength and intensity must be specified)
    wavelength = 640E-9
    I = 6E15 # Intensity in photons/m^2/pulse

    # If genProfile is 'none', specify the generation rate G_ehp (this will be applied to the layer specified by lyr_idx)
    G_ehp = 1E31 # in m^-3 s^-1

    ########### Misc. pars ###########

    # UUID = str(uuid.uuid4()) # Add a UUID to the simulation
    UUID = ''
    run_mode = False  # If False, show verbose output in console

    ############## Command line arguments  ##############
    ## Notes
    ## - The command line arguments are optional and can be provided in any order
    ## - Each command line argument must be provided in the format -par_name value
    ## - Possible arguments include all SIMsalabim parameters and trPL specific parameters as listed before
    ## - Special arguments
    ##   - sp : string
    ##     - The session path, i.e. the working directory for the simulation
    ##   - simsetup : string
    ##     - The name of the zimt simulation setup parameters file
    ##   - UUID : string
    ##     - An UUID to add to the simulation (output)

    cmd_pars_dict = {}
    cmd_pars = [] # When specifying command line arguments for a contactless device, use the layer indices as defined in the original device architecture. The function will automatically adjust the layer indices needed.

    # Check if any arguments are provided.
    if len(sys.argv) >= 2:
        # Each arguments should be in the format -par val, i.e. in pairs of two. Skip the first argument as this is the script name
        if not len(sys.argv[1:]) % 2 == 0:
            print('Error in command line parameters. Please provide arguments in the format -par_name value -par_name value ...')
            sys.exit(1)
        else:
            input_list = sys.argv[1:]
            # Loop over the input list and put pairs into a dictionary with the first argument as key and the second argument as value
            for i in range(0,len(input_list),2):
                # Check if the key already exists, if not, add it to the cmd_pars_dict
                if str(input_list[i][1:]) in cmd_pars_dict:
                    print(f'Duplicate parameter found in the command line parameters: {str(input_list[i][1:])}')
                    sys.exit(1)
                else:
                    cmd_pars_dict[str(input_list[i][1:])] = str(input_list[i+1])

    # Check and process specific keys/arguments that are not native SIMsalabim arguments
    # Handle the session_path/sp argument separately, as the other parameters depend on this
    if 'sp' in cmd_pars_dict:
        session_path = cmd_pars_dict['sp']
        # remove from cmd_pars
        cmd_pars_dict.pop('sp')

    # Define mappings for keys and variables
    key_action_map = {
        'simsetup': lambda val: {'zimt_device_parameters': val},
        't_max': lambda val: {'t_max': float(val)},
        't_step': lambda val: {'t_step': float(val)},
        't_pulse': lambda val: {'t_pulse': int(val)},
        'FWHM': lambda val: {'FWHM': float(val)},
        'spectrum_type': lambda val: {'spectrum_type': val},
        'I': lambda val: {'I': float(val)},
        'wavelength': lambda val: {'wavelength': float(val)},
        'lyr_idx': lambda val: {'lyr_idx': int(val)},
        'tVG_name': lambda val: {'tVG_name': val},
        'tj_name': lambda val: {'tj_name': val},
        'out_name': lambda val: {'output_name': val},
        'UUID': lambda val: {'UUID': val},
    }

    for key in list(cmd_pars_dict.keys()):  # Use list to avoid modifying the dictionary while iterating
        if key in key_action_map:
            # Apply the corresponding action
            result = key_action_map[key](cmd_pars_dict[key])
            globals().update(result)  # Dynamically update global variables
            cmd_pars_dict.pop(key)

    # Handle remaining keys in `cmd_pars_dict` and add them to the cmd_pars list
    cmd_pars.extend({'par': key, 'val': value} for key, value in cmd_pars_dict.items())

    result, message = run_trPL_simu(simulation_setup_zimt,session_path, lyr_idx, t_max, t_scale, t_step, t_pulse, FWHM, 
                                    genProfile, spectrum_type, I, wavelength, G_ehp, contactless_device=contactless_device, format_output=format_output, run_mode=False, 
                                    tVG_name=tVG_name, output_file = output_name, tj_name = tj_name, cmd_pars=cmd_pars, UUID=UUID)
    # Make the trPL figure
    if result == 0 or result == 95:

        # Custom function to get the x and y bounds
        xlim = [t_pulse - 0.5*t_pulse, t_max] # Force x-axis limits for better visibility of the actual trPL signal on a log scale

        data_trPL = pd.read_csv(os.path.join(session_path, output_name),sep=r'\s+')
        y_min = data_trPL[data_trPL['t'] > t_pulse]['trPL'].min()/2
        y_max = data_trPL['trPL'].max()*2

        plot_trPL(session_path, os.path.basename(output_name), xlim, [y_min, y_max])
    else:
        print(result, message)
        sys.exit(1)
